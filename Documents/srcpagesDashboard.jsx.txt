import { useEffect, useState } from "react";
import { fetchDashboardData } from "../../services/dashboardService";

export default function Dashboard() {
  const [data, setData] = useState([]);

  useEffect(() => {
    loadData();
  }, []);

  async function loadData() {
    const result = await fetchDashboardData();
    setData(result);
  }

  // 🔢 KPIs
  const totalExpected = data.reduce((sum, r) => sum + Number(r.total_billed), 0);
  const totalPaid = data.reduce((sum, r) => sum + Number(r.total_paid), 0);
  const totalBalance = data.reduce((sum, r) => sum + Number(r.total_balance), 0);

  const occupancy =
    data.length === 0 ? 0 : ((data.length / data.length) * 100).toFixed(0);

  const defaulters = data.filter((r) => r.total_balance > 0);

  return (
    <div style={{ padding: 20 }}>
      <h2>Dashboard</h2>

      {/* KPI CARDS */}
      <div style={{ display: "flex", gap: 20, marginBottom: 20 }}>
        <Card title="Expected Rent" value={totalExpected} />
        <Card title="Collected" value={totalPaid} />
        <Card title="Outstanding" value={totalBalance} />
        <Card title="Occupancy %" value={occupancy + "%"} />
      </div>

      {/* RENT ROLL */}
      <h3>Rent Roll</h3>
      <table border="1" cellPadding="10">
        <thead>
          <tr>
            <th>Tenant</th>
            <th>Unit</th>
            <th>Rent</th>
            <th>Billed</th>
            <th>Paid</th>
            <th>Balance</th>
            <th>Status</th>
          </tr>
        </thead>
        <tbody>
          {data.map((r) => (
            <tr key={r.lease_id}>
              <td>{r.tenant_name}</td>
              <td>{r.unit_code}</td>
              <td>{r.rent_amount}</td>
              <td>{r.total_billed}</td>
              <td>{r.total_paid}</td>
              <td>{r.total_balance}</td>
              <td>{r.status}</td>
            </tr>
          ))}
        </tbody>
      </table>

      {/* DEFAULTERS */}
      <h3 style={{ marginTop: 30 }}>Defaulters</h3>
      <table border="1" cellPadding="10">
        <thead>
          <tr>
            <th>Tenant</th>
            <th>Unit</th>
            <th>Balance</th>
          </tr>
        </thead>
        <tbody>
          {defaulters.map((r) => (
            <tr key={r.lease_id}>
              <td>{r.tenant_name}</td>
              <td>{r.unit_code}</td>
              <td style={{ color: "red" }}>{r.total_balance}</td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}

function Card({ title, value }) {
  return (
    <div
      style={{
        padding: 20,
        border: "1px solid #ddd",
        borderRadius: 10,
        minWidth: 150,
      }}
    >
      <h4>{title}</h4>
      <h2>{value}</h2>
    </div>
  );
}
