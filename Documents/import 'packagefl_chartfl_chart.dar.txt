class KioskSummary {
  final String id;
  final String name;
  final String location;
  final bool isActive;
  final String staffAssigned;
  final String lastReading;
  final String reconciliationStatus;

  KioskSummary({
    required this.id,
    required this.name,
    required this.location,
    required this.isActive,
    required this.staffAssigned,
    required this.lastReading,
    required this.reconciliationStatus,
  });
}

class DailyPoint {
  final DateTime date;
  final double value;

  DailyPoint({required this.date, required this.value});
}

class OwnerAnalytics {
  final List<DailyPoint> water;
  final List<DailyPoint> revenue;
  final List<DailyPoint> variance;

  OwnerAnalytics({
    required this.water,
    required this.revenue,
    required this.variance,
  });
}

// ---------------------------------------------------------
// OWNER DASHBOARD SCREEN
// ---------------------------------------------------------

class OwnerDashboardScreen extends StatefulWidget {
  final String ownerId;

  const OwnerDashboardScreen({
    super.key,
    required this.ownerId,
  });

  @override
  State<OwnerDashboardScreen> createState() => _OwnerDashboardScreenState();
}

class _OwnerDashboardScreenState extends State<OwnerDashboardScreen> {
  final supabase = Supabase.instance.client;

  late Future<Map<String, dynamic>> _metricsFuture;
  late Future<OwnerAnalytics> _analyticsFuture;
  late Future<List<KioskSummary>> _kiosksFuture;

  @override
  void initState() {
    super.initState();
    _metricsFuture = _loadMetrics();
    _analyticsFuture = _loadAnalytics();
    _kiosksFuture = _loadKiosks();
  }

  String get _todayDateString =>
      DateTime.now().toIso8601String().substring(0, 10);

  String get _yesterdayDateString =>
      DateTime.now()
          .subtract(const Duration(days: 1))
          .toIso8601String()
          .substring(0, 10);

  // ---------------------------------------------------------
  // METRICS LOADER
  // ---------------------------------------------------------

  Future<Map<String, dynamic>> _loadMetrics() async {
    try {
      final today = _todayDateString;
      final yesterday = _yesterdayDateString;

      // STEP 1: get kiosk_ids from k['id'] where k['owner_id'] = widget.ownerId
      final kiosks = await supabase
          .schema('water')
          .from('kiosks')
          .select('id')
          .eq('owner_id', widget.ownerId);

      debugPrint("RAW KIOSKS RESPONSE: $kiosks");

      if ((kiosks as List).isEmpty) {
        return {
          'totalKiosks': 0,
          'todayRevenue': 0.0,
          'yesterdayRevenue': 0.0,
          'todayWater': 0.0,
          'yesterdayWater': 0.0,
        };
      }

      final kioskIds = (kiosks as List)
          .map<String>((k) => k['id']?.toString() ?? '')
          .where((id) => id.isNotEmpty)
          .toList();

      if (kioskIds.isEmpty) {
        return {
          'totalKiosks': 0,
          'todayRevenue': 0.0,
          'yesterdayRevenue': 0.0,
          'todayWater': 0.0,
          'yesterdayWater': 0.0,
        };
      }

      // ---------------------------
      // TODAY DATA
      // ---------------------------
      final todayRows = await supabase
          .schema('water')
          .from('daily_kiosk_summary')
          .select()
          .inFilter('kiosk_id', kioskIds)
          .eq('date', today);

      // ---------------------------
      // FALLBACK TO LATEST DATE
      // ---------------------------
      List rowsToUse = todayRows;

      if ((todayRows as List).isEmpty) {
        debugPrint("⚠️ No data for today, fetching latest available date");

        final latestRows = await supabase
            .schema('water')
            .from('daily_kiosk_summary')
            .select()
            .inFilter('kiosk_id', kioskIds)
            .order('date', ascending: false);

      rowsToUse = latestRows;
      }      

      double todayWater = 0;
      double todayRevenue = 0;

      for (final row in rowsToUse as List) {
        todayWater += (row['total_liters'] ?? 0).toDouble();
        todayRevenue += (row['total_revenue'] ?? 0).toDouble();
      }

      // ---------------------------
      // YESTERDAY DATA (unchanged)
      // ---------------------------
      final yesterdayRows = await supabase
          .schema('water')
          .from('daily_kiosk_summary')
          .select()
          .inFilter('kiosk_id', kioskIds)
          .eq('date', yesterday);

      double yesterdayWater = 0;
      double yesterdayRevenue = 0;

      for (final row in yesterdayRows as List) {
        yesterdayWater += (row['total_liters'] ?? 0).toDouble();
        yesterdayRevenue += (row['total_revenue'] ?? 0).toDouble();
      }

      return {
        'totalKiosks': kioskIds.length,
        'todayRevenue': todayRevenue,
        'yesterdayRevenue': yesterdayRevenue,
        'todayWater': todayWater,
        'yesterdayWater': yesterdayWater,
      };
    } catch (e) {
      debugPrint("🚨 METRICS ERROR: $e");

      return {
        'totalKiosks': 0,
        'todayRevenue': 0.0,
        'yesterdayRevenue': 0.0,
        'todayWater': 0.0,
        'yesterdayWater': 0.0,
      };
    }
  }
  // ---------------------------------------------------------
  // ANALYTICS LOADER
  // ---------------------------------------------------------

  Future<OwnerAnalytics> _loadAnalytics() async {
    try {
      final startDate = DateTime.now()
          .subtract(const Duration(days: 6))
          .toIso8601String()
          .substring(0, 10);

      final today = _todayDateString;

      final kiosks = await supabase
          .schema('water')
          .from('kiosks')
          .select('id')
          .eq('owner_id', widget.ownerId);

      final kioskIds =
          (kiosks as List).map<String>((k) => k['id'] as String).toList();

      if (kioskIds.isEmpty) {
        return OwnerAnalytics(water: [], revenue: [], variance: []);
      }

      final rows = await supabase
          .schema('water')
          .from('daily_kiosk_summary')
          .select()
          .inFilter('kiosk_id', kioskIds)
          .gte('date', startDate)
          .lte('date', today)
          .order('date');

      final Map<String, double> water = {};
      final Map<String, double> revenue = {};
      final Map<String, double> variance = {};

      for (final row in rows as List) {
       final d = row['date'];

      water[d] = (water[d] ?? 0) + (row['total_liters'] ?? 0).toDouble();

      revenue[d] =
          (revenue[d] ?? 0) + (row['total_revenue'] ?? 0).toDouble();

      // optional (since no column exists)
      variance[d] = 0;
      }

      List<DailyPoint> build(Map<String, double> map) {
        final keys = map.keys.toList()..sort();
        return [
          for (final k in keys)
            DailyPoint(date: DateTime.parse(k), value: map[k] ?? 0),
        ];
      }

      return OwnerAnalytics(
        water: build(water),
        revenue: build(revenue),
        variance: build(variance),
      );
    } catch (e) {
      return OwnerAnalytics(water: [], revenue: [], variance: []);
    }
  }

  // ---------------------------------------------------------
  // KIOSK LOADER
  // ---------------------------------------------------------

  Future<List<KioskSummary>> _loadKiosks() async {
  // STEP 1: get kiosk_ids for owner
  final kiosks = await supabase
      .schema('water')
      .from('kiosks')
      .select('id')
      .eq('owner_id', widget.ownerId);

    debugPrint("RAW KIOSKS RESPONSE (KIOSK LOADER): $kiosks");

  if ((kiosks as List).isEmpty) {
    return [];
    }

  final kioskIds = (kiosks as List)
      .map<String>((k) => k['id'] as String)
      .toList();

    debugPrint("DEBUG STEP 1 ----------------");
    debugPrint("OWNER ID: ${widget.ownerId}");
    

    debugPrint("DEBUG STEP 2 ----------------");
    debugPrint("KIOSK IDS: $kioskIds");  

  if (kioskIds.isEmpty) return [];

  // ---------------------------
  // BULK: LAST READINGS
  // ---------------------------
  final readings = await supabase
      .schema('water')
      .from('meter_readings')
      .select('kiosk_id, closing_reading, created_at')
      .inFilter('kiosk_id', kioskIds)
      .order('created_at', ascending: false);

  // ---------------------------
  // BULK: TODAY SUMMARIES
  // ---------------------------
  final today = DateTime.now().toIso8601String().substring(0, 10);

  final summaries = await supabase
      .schema('water')
      .from('daily_kiosk_summary')
      .select('kiosk_id')
      .inFilter('kiosk_id', kioskIds)
      .eq('date', today);

   // ---------------------------
  // BUILD LOOKUPS
  // ---------------------------
  final Map<String, dynamic> lastReadingMap = {};

  for (final r in readings as List) {
    final kid = r['kiosk_id'];
    if (!lastReadingMap.containsKey(kid)) {
      lastReadingMap[kid] = r;
    }
  }

  final Set<String> submittedKiosks =
      (summaries as List).map((s) => s['kiosk_id'] as String).toSet();   

  // STEP 2: fetch kiosk details
  final rows = await supabase
      .schema('water')
      .from('kiosks')
      .select('''
        id,
        name,
        location,
        active,
        user_kiosks (
          role_at_kiosk,
          users (
            id,
            name,
            phone
          )
        )
      ''')
      .inFilter('id', kioskIds)
      .order('name');

    debugPrint("DEBUG STEP 3 ----------------");
    debugPrint("KIOSK ROWS: $rows"); 

  final enriched = <KioskSummary>[];

  for (final k in rows as List) {
  debugPrint("PROCESSING KIOSK: $k");
  final kioskId = k['id'];

  // ---------------------------
  // 1. LAST READING (from map)
  // ---------------------------
  final lastReadingRes = lastReadingMap[kioskId];

  final lastReading = lastReadingRes != null
      ? "${lastReadingRes['closing_reading']} L"
      : "No readings";

  // ---------------------------
  // 2. STATUS (from set)
  // ---------------------------
  final status = submittedKiosks.contains(kioskId)
      ? 'approved'
      : 'pending';

  // ---------------------------
  // 3. STAFF
  // ---------------------------
  final staffList = k['user_kiosks'] as List?;

  final assigned = (staffList != null && staffList.isNotEmpty)
      ? staffList.map((s) {
          final user = s['users'];
          final role = s['role_at_kiosk'] ?? 'staff';

          final name = (user != null && user['name'] != null)
              ? user['name']
              : 'Unknown';

          return "$name ($role)";
        }).join(", ")
      : "Unassigned";

    enriched.add(
      KioskSummary(
        id: kioskId,
        name: k['name'] ?? 'Unnamed Kiosk',
        location: k['location'] ?? 'Unknown',
        isActive: k['active'] ?? false,
        staffAssigned: assigned,
        lastReading: lastReading,
        reconciliationStatus: status,
      ),
    );
  }

  return enriched;
  }

  // ---------------------------------------------------------
  // TREND BADGE
  // ---------------------------------------------------------

  Widget buildTrendBadge({
    required double today,
    required double? yesterday,
  }) {
    if (yesterday == null || yesterday == 0) {
      return _trendBadge("N/A", Colors.grey);
    }

    final diff = today - yesterday;
    final percent = ((diff / yesterday) * 100).round();

    if (diff > 0) {
      return _trendBadge("↑ $percent%", Colors.green);
    } else if (diff < 0) {
      return _trendBadge("↓ ${percent.abs()}%", Colors.red);
    } else {
      return _trendBadge("→ 0%", Colors.grey);
    }
  }

  Widget _trendBadge(String text, Color color) {
    return Container(
      padding: const EdgeInsets.symmetric(horizontal: 6, vertical: 2),
      decoration: BoxDecoration(
        color: color.withOpacity(0.12),
        borderRadius: BorderRadius.circular(8),
      ),
      child: Text(
        text,
        style: TextStyle(
          fontSize: 11,
          fontWeight: FontWeight.w600,
          color: color,
        ),
      ),
    );
  }

  // ---------------------------------------------------------
  // BUILD METHOD
  // ---------------------------------------------------------

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Owner Dashboard'),
      ),
      
      body: SingleChildScrollView(
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Padding(
              padding: const EdgeInsets.all(16.0),
              child: Breadcrumbs(
                items: [
                  BreadcrumbItem(label: "Dashboard"),
                ],
              ),
            ),
            const SizedBox(height: 4),

            // METRICS
            FutureBuilder<Map<String, dynamic>>(
  future: _metricsFuture,
  builder: (context, snapshot) {
    if (snapshot.connectionState == ConnectionState.waiting) {
      return const Padding(
        padding: EdgeInsets.all(16),
        child: Center(child: CircularProgressIndicator()),
      );
    }

    if (snapshot.hasError) {
      return const Padding(
        padding: EdgeInsets.all(16),
        child: Text('Failed to load metrics'),
      );
    }

    if (!snapshot.hasData) {
      return const SizedBox.shrink();
    }

    // ✅ THIS LINE WAS LOST — RESTORE IT
    final m = snapshot.data!;

    return Padding(
      padding: const EdgeInsets.all(16),
      child: Wrap(
        spacing: 12,
        runSpacing: 12,
        children: [
          _MetricCard(
            icon: Icons.store,
            label: 'Total Kiosks',
            value: '${m['totalKiosks']}',
            color: const Color(0xFFE3F2FD),
          ),
          _MetricCard(
            icon: Icons.payments,
            label: 'Today\'s Revenue',
            value: 'KSH ${(m['todayRevenue'] as num).round()}',
            color: const Color(0xFFE8F5E9),
            trailing: buildTrendBadge(
              today: (m['todayRevenue'] as num).toDouble(),
              yesterday: (m['yesterdayRevenue'] as num).toDouble(),
            ),
          ),
          _MetricCard(
            icon: Icons.water_drop,
            label: 'Today\'s Water',
            value: '${(m['todayWater'] as num).round()} L',
            color: const Color(0xFFE0F7FA),
            trailing: buildTrendBadge(
              today: (m['todayWater'] as num).toDouble(),
              yesterday: (m['yesterdayWater'] as num).toDouble(),
            ),
          ),
        ],
      ),
    );
  },
),

            // ANALYTICS
            FutureBuilder<OwnerAnalytics>(
              future: _analyticsFuture,
              builder: (context, snapshot) {
                if (snapshot.connectionState == ConnectionState.waiting) {
                  return const Padding(
                    padding: EdgeInsets.all(16),
                    child: Center(child: CircularProgressIndicator()),
                  );
                }

                if (snapshot.hasError) {
                  return const Padding(
                    padding: EdgeInsets.all(16),
                    child: Text('Failed to load analytics'),
                  );
                }

                if (!snapshot.hasData) {
                  return const SizedBox.shrink();
                }

                final a = snapshot.data!;
                return SizedBox(
                  height: 220,
                  child: SingleChildScrollView(
                    scrollDirection: Axis.horizontal,
                    padding: const EdgeInsets.symmetric(horizontal: 16),
                    child: Row(
                      children: [
                        _AnalyticsCard(
                          title: 'Water (Last 7 Days)',
                          child: WaterChart(data: a.water),
                        ),
                        const SizedBox(width: 12),
                        _AnalyticsCard(
                          title: 'Revenue (Last 7 Days)',
                          child: RevenueChart(data: a.revenue),
                        ),
                        const SizedBox(width: 12),
                        _AnalyticsCard(
                          title: 'Variance (Last 7 Days)',
                          child: VarianceChart(data: a.variance),
                        ),
                      ],
                    ),
                  ),
                );
              },
            ),

            // KIOSKS
            const Padding(
              padding: EdgeInsets.symmetric(horizontal: 16),
              child: Text(
                'Your Kiosks',
                style: TextStyle(
                  fontSize: 18,
                  fontWeight: FontWeight.w600,
                ),
              ),
            ),

            SizedBox(
  height: 400,
  child: FutureBuilder<List<KioskSummary>>(
    future: _kiosksFuture,
    builder: (context, snapshot) {
      // ---------------------------
      // DEBUG LOGS
      // ---------------------------
      debugPrint("📡 STATE: ${snapshot.connectionState}");
      debugPrint("📦 DATA: ${snapshot.data}");
      debugPrint("❌ ERROR: ${snapshot.error}");

      // ---------------------------
      // LOADING
      // ---------------------------
      if (snapshot.connectionState == ConnectionState.waiting) {
        return const Center(child: CircularProgressIndicator());
      }

      // ---------------------------
      // ERROR
      // ---------------------------
      if (snapshot.hasError) {
        debugPrint("🚨 KIOSK ERROR: ${snapshot.error}");
        return const Center(child: Text("Error loading kiosks"));
      }

      // ---------------------------
      // NULL DATA
      // ---------------------------
      final kiosks = snapshot.data;

      if (kiosks == null) {
        debugPrint("⚠️ SNAPSHOT DATA IS NULL");
        return const Center(child: Text("No data returned"));
      }

      // ---------------------------
      // EMPTY LIST
      // ---------------------------
      if (kiosks.isEmpty) {
        debugPrint("⚠️ KIOSK LIST EMPTY");
        return const Center(child: Text("No kiosks found"));
      }

      // ---------------------------
      // SUCCESS
      // ---------------------------
      debugPrint("✅ KIOSKS READY: ${kiosks.length}");

      return ListView.builder(
        padding: const EdgeInsets.symmetric(horizontal: 16),
        itemCount: kiosks.length,
        itemBuilder: (context, i) {
          final k = kiosks[i];

          debugPrint("🏪 BUILDING: ${k.name}");

          return Padding(
            padding: const EdgeInsets.only(bottom: 12),
            child: _KioskCard(
              kiosk: k,
              onOpenKioskDashboard: () {
                debugPrint("🔥 STEP A: Button tapped");

                final route = '/owner/kiosk-dashboard';
                debugPrint("➡️ STEP B: About to navigate to $route");

  try {
    GoRouter.of(context).push(
      route,
      extra: {
        'id': k.id,
        'name': k.name,
      },
    );
    debugPrint("✅ STEP C: push() called");
  } catch (e) {
    debugPrint("🚨 STEP D: Navigation failed: $e");
  }
},
            ),
          );
        },
      );
    },
  ),
)
          ],
        ),
      ),
    );
  }
}

class WaterChart extends StatelessWidget {
  final List<DailyPoint> data;

  const WaterChart({required this.data});

  @override
  Widget build(BuildContext context) {
    if (data.isEmpty) return const Center(child: Text('No data'));

    return LineChart(
      LineChartData(
        minX: 0,
        maxX: (data.length - 1).toDouble(),
        minY: 0,

        gridData: FlGridData(show: true),
        borderData: FlBorderData(show: true),
        titlesData: FlTitlesData(show: true),

        lineBarsData: [
          LineChartBarData(
            spots: [
              for (int i = 0; i < data.length; i++)
                FlSpot(i.toDouble(), data[i].value),
            ],
            isCurved: true,
            color: Colors.blue,
            barWidth: 3,
            dotData: FlDotData(show: true),
          ),
        ],
      ),
    );
  }
}
